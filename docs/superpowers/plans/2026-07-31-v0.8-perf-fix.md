# v0.8 紧急性能修复 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让混淆后脚本在 Roblox 脚本超时内正常加载执行——压缩 keyfuse dispatch 状态机、移除解密链中的 LZW、降低不透明谓词频率。

**Architecture:**
- keyfuse：每个 dispatch case 装配 4 个 nibble（128→32 真实 case + 11 decoy = 43 状态）。`_kh` 宿主表保持 128 真 + 44 假条目不变，故 `computeKeyfuseKhSize` / rtToken 不变，rtdeps 路径零回归。
- packer + 运行时模板：移除 LZW 压缩/解压，仅保留 stream XOR（+ keyfuse xor512 / rt_mix）。
- flatten + deadcode：不透明谓词概率 0.5 → 0.12（flatten），0.18–0.28 → 0.08–0.16（deadcode）。

**Tech Stack:** TypeScript (vitest), Luau 运行时模板（runtime/vm-runtime.template.lua）。

**已确认的规格参数（来自用户澄清）:**
- 切分粒度：4 nibble/分支（2 字节）→ 32 真实 case + 11 decoy ≈ 43 状态（命中 30–50）。
- bi 层范围：全局移除 LZW，保留 stream XOR（keyfuse 与非 keyfuse 路径均移除）。
- 谓词频率：p ≈ 0.12（约每 8 分支一次）。

**沙箱限制:** 无 luau 二进制（`command -v luau` → NO_LUAU）。Roblox 5 秒端到端验证为文档化手工步骤；自动化测试用结构断言（case 数 ≤ 50、无 `lzw_decode`、500 行脚本混淆完成）替代。

- [ ] **Step 2: 写失败测试——重写 simulateAssembly 支持 4-hex**

替换 `tests/keyfuse.test.ts` 中整个 `simulateAssembly` 函数为：

```ts
/**
 * 解析 genKeyfuseAssembly 生成的 assemblyCode 并模拟 dispatch loop 执行，
 * 验证装配出的 KEY == keyHex。v0.8：每个 case 装配 4 个 nibble（4 hex 字符）。
 */
function simulateAssembly(keyHex: string, seed: number): string {
  const kf = genKeyfuseAssembly(keyHex, seed);
  const code = kf.assemblyCode;

  // 解析 _kh 表。
  const khMatch = code.match(/local _kh = \{([\s\S]*?)\}/);
  expect(khMatch).not.toBeNull();
  const khValues: number[] = khMatch![1]
    .split(",")
    .map((s) => s.trim())
    .filter((s) => s.length > 0)
    .map((s) => Number(s));

  // 解析 _kk。
  const kkMatch = code.match(/local _kk = (\d+)/);
  expect(kkMatch).not.toBeNull();
  const kk = Number(kkMatch![1]);

  // 解析初始 state。
  const initMatch = code.match(/local __kf_b = (\d+)/);
  expect(initMatch).not.toBeNull();
  let curState = Number(initMatch![1]);

  // 解析所有 case：stateId → (args 表达式列表, nextState)。
  // 每个 case 形如：__kf_b == SID then KEY = KEY .. fmt("%X%X%X%X", arg0, arg1, arg2, arg3) __kf_b = NEXT
  const caseRe =
    /__kf_b == (\d+) then\s*\n\s*KEY = KEY \.\. string\.format\("%X%X%X%X", ([^\n]+)\s*\n\s*__kf_b = (\d+)/g;
  const exitRe = /__kf_b == (\d+) then\s*\n\s*break/;
  const cases = new Map<number, { args: string[]; next: number }>();
  let m: RegExpExecArray | null;
  while ((m = caseRe.exec(code)) !== null) {
    const args = m[2]!.split(",").map((s) => s.trim());
    cases.set(Number(m[1]), { args, next: Number(m[3]) });
  }
  const exitMatch = code.match(exitRe);
  expect(exitMatch).not.toBeNull();
  const exitState = Number(exitMatch![1]);

  // 模拟 dispatch loop。每 case 4 个 arg，每个 arg 是 _kh[_B(i,_kk)+1]%16。
  let key = "";
  let steps = 0;
  const visited = new Set<number>();
  while (curState !== exitState) {
    if (visited.has(curState)) {
      throw new Error(`simulate: state ${curState} revisited (cycle)`);
    }
    visited.add(curState);
    const c = cases.get(curState);
    if (!c) {
      throw new Error(`simulate: no case for state ${curState}`);
    }
    for (const arg of c.args) {
      const am = arg.match(/^_kh\[_B\((\d+), _kk\) \+ 1\] % 16$/);
      if (!am) {
        throw new Error(`simulate: unexpected arg ${arg} (rtToken path not exercised here)`);
      }
      const i = Number(am[1]);
      const slot = _B(i, kk); // 0-based
      const nibble = khValues[slot]! % 16;
      key += nibble.toString(16).toUpperCase();
    }
    curState = c.next;
    if (++steps > 10000) throw new Error("simulate: too many steps");
  }
  return key;
}
```

- [ ] **Step 6: 实现——rework case 发射为 4-hex**

在 `src/vm/keyfuse.ts` 中，替换 `for (const caseIdx of caseOrder) { ... }` 整个循环体（从 `if (caseIdx < numRealCases) {` 到对应 `}` 的三分支 if/elseif/else 结束）为：

```ts
  for (const caseIdx of caseOrder) {
    if (caseIdx < numRealCases) {
      // 真实 case g：装配 nibbles[4g .. 4g+3]（4 nibble = 2 字节 / 分支）。
      const g = caseIdx;
      const baseIdx = g * NIBBLES_PER_CASE;
      const sid = stateIds[g]!;
      const nextSid = g + 1 < numRealCases ? stateIds[g + 1]! : exitId;
      const branchKw = firstCase ? "if" : "elseif";
      firstCase = false;
      lines.push(`  ${branchKw} __kf_b == ${sid} then`);
      const args: string[] = [];
      for (let k = 0; k < NIBBLES_PER_CASE; k++) {
        const j = baseIdx + k;
        if (rtActive && (j === 126 || j === 127)) {
          // v0.10 rt_deps：nibbles 126/127 从 _rt_tok 派生。
          const shift = j === 126 ? 4 : 8;
          args.push(`math.floor(_rt_tok / ${1 << shift}) % 16`);
        } else {
          args.push(`_kh[_B(${j}, _kk) + 1] % 16`);
        }
      }
      const fmt = "%X".repeat(NIBBLES_PER_CASE);
      lines.push(`    KEY = KEY .. string.format("${fmt}", ${args.join(", ")})`);
      lines.push(`    __kf_b = ${nextSid}`);
    } else if (caseIdx < totalCases) {
      // 死分支 case：结构与真实 case 完全一致（4 hex 字符），循环引用 decoy 宿主槽。
      const decoyPos = caseIdx - numRealCases;
      const sid = stateIds[caseIdx]!;
      const fakeNext = stateIds[caseIdx]!; // 自指（不可达，无意义）
      const branchKw = firstCase ? "if" : "elseif";
      firstCase = false;
      lines.push(`  ${branchKw} __kf_b == ${sid} then`);
      const args: string[] = [];
      for (let k = 0; k < NIBBLES_PER_CASE; k++) {
        // 循环遍历 decoy 宿主槽（129..128+decoyHostIdxs.length）。
        const fakeSlot = 129 + ((decoyPos * NIBBLES_PER_CASE + k) % decoyHostIdxs.length);
        const fakeI = (fakeSlot - 1) ^ kk; // 反推 i 使 _B(i,_kk)+1 == fakeSlot
        args.push(`_kh[_B(${fakeI}, _kk) + 1] % 16`);
      }
      const fmt = "%X".repeat(NIBBLES_PER_CASE);
      lines.push(`    KEY = KEY .. string.format("${fmt}", ${args.join(", ")})`);
      lines.push(`    __kf_b = ${fakeNext}`);
    } else {
      // exit case。
      const branchKw = firstCase ? "if" : "elseif";
      firstCase = false;
      lines.push(`  ${branchKw} __kf_b == ${exitId} then`);
      lines.push(`    break`);
    }
  }
```

```ts
/** 生成 ~500 行 SANA HUB 风格 Lua（多函数 + 表 + 循环 + 字符串）。 */
function generateSanaHub500(): string {
  const lines: string[] = [];
  lines.push('local SANA_HUB = {}');
  lines.push('SANA_HUB.Modules = {}');
  lines.push('SANA_HUB.Config = { Theme = "Dark", Toggle = true }');
  for (let i = 0; i < 40; i++) {
    lines.push(`local function mod${i}(x)`);
    lines.push(`  local s = "module_${i}_data"`);
    lines.push(`  local acc = 0`);
    lines.push(`  for k = 1, 10 do`);
    lines.push(`    acc = acc + k * (x + ${i})`);
    lines.push(`  end`);
    lines.push(`  SANA_HUB.Modules[${i}] = { name = s, value = acc }`);
    lines.push(`  return acc`);
    lines.push(`end`);
  }
  for (let i = 0; i < 40; i++) {
    lines.push(`local r${i} = mod${i}(${i + 1})`);
  }
  lines.push('local function summarize()');
  lines.push('  local total = 0');
  lines.push('  for idx, m in pairs(SANA_HUB.Modules) do');
  lines.push('    total = total + m.value');
  lines.push('  end');
  lines.push('  return total');
  lines.push('end');
  lines.push('SANA_HUB.Sum = summarize()');
  lines.push('print("SANA_HUB loaded: " .. tostring(SANA_HUB.Sum))');
  // 填充到 >= 500 行
  while (lines.length < 500) {
    lines.push(`local _pad${lines.length} = ${lines.length} * 2 + 1`);
  }
  return lines.join("\n");
}

describe("v0.8 performance: SANA HUB 500-line", () => {
  const src = generateSanaHub500();

  it("benchmark source is >= 500 lines", () => {
    expect(src.split("\n").length).toBeGreaterThanOrEqual(500);
  });

  it("keyfuse dispatch cases <= 50 and >= 30 (pre-self-obf runtime)", () => {
    const ast = parse(lex(src));
    const rt = compileVMWithRuntime(ast, 42, DEFAULT_RUNTIME_PROTECT, "f6");
    const caseCount = (rt.match(/__kf_b ==/g) || []).length;
    expect(caseCount).toBeGreaterThanOrEqual(30);
    expect(caseCount).toBeLessThanOrEqual(50);
  });

  it("no lzw_decode in pre-self-obf runtime", () => {
    const ast = parse(lex(src));
    const rt = compileVMWithRuntime(ast, 42, DEFAULT_RUNTIME_PROTECT, "f6");
    expect(rt).not.toContain("lzw_decode");
  });

  it("full pipeline (vm + runtime) completes without crash", () => {
    const { out } = runPipeline(src, { seed: 42, vm: true, runtime: true });
    expect(out.length).toBeGreaterThan(0);
    expect(out).toContain("CUA混淆器");
  });
});
```
